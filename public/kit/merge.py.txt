In the site's build folder, after build.py has made dist/:

    python3 merge.py                      -> dist/index.html and dist/llms.txt, merged in place

The page as build.py made it is kept beside it as dist/index.before-merge.html, and a second run starts from
that copy. The plugin area comes from the folder plugins/ beside this script: the plugin page as built
(index.html, or index.html.txt as the kit keeps it), its stylesheet (entry.css) and its script (app.js).
To try a change before it goes to the site:

    python3 merge.py --site <page> --plugins <folder> --out <page> --preview

--preview adds the lime strip that marks a staging copy and gives the page the staging copy's title.

It works on built pages, not on sources, and changes neither input.

What the merge does:
1. Navigation. Plugins becomes an in-page section of the site's own navigation.
2. Plugin area. Its eleven views go inside one wrapper (.plg) at the end of <main>, with their own second row
   (a way back, the plugin's name, its sections), shown only while a plugin view is open.
3. Styles. The plugin area's stylesheet is scoped to the wrapper so that it cannot restyle the site's views, and
   nine class names that mean something else in the site's stylesheet are renamed inside the plugin area.
4. Script. One router answers every change of address. The site's script still opens the first view itself,
   and the plugin area's router marks the site's navigation.
5. Film. The 60-second film is left out of Configure, at the owner's request (FILM = False).
6. Home. The opener leads with a finding from the register. The diagram of agents and people moves down, to
   sit under "Our read" as its illustration.
7. llms.txt. The plugin pages are added to the site's list for AI readers.

def drop_film(h):
    """Take the 60-second film out of the Configure opener, which goes back to one column like the other openers."""
    a = h.index('<figure class="film" id="film">')
    b = h.index("</figure>", a) + len("</figure>")

# ----------------------------------------------------------------------------- the plugin area's styles
def split_selectors(s):
    out, depth, cur = [], 0, ""
    for ch in s:
        if ch in "([":
            depth += 1
        elif ch in ")]":
            depth -= 1
        if ch == "," and depth == 0:
            out.append(cur)
            cur = ""
        else:
            cur += ch
    out.append(cur)
    return [x.strip() for x in out if x.strip()]


DROP_PREFIX = (".mast .", ".wm", ".asof", ".foot")


def scope_selector(sel):
    """Tie one selector to the plugin area, or return None to drop a rule that restyled the masthead or footer.
    :where() adds no weight, so every rule keeps the rank it had against the site's styles on the plugin page."""
    if sel == "html":
        return "html.plg-on"
    if sel == ".mast":
        return ".plg-on .mast"
    if sel.startswith(DROP_PREFIX):
        return None
    if sel.startswith("html:not(.js)"):
        return "html:not(.js) :where(.plg)" + sel[len("html:not(.js)"):]
    if sel.startswith(".js "):
        return ".js :where(.plg) " + sel[4:]
    return ":where(.plg) " + sel


# Rules that exist only in the site's stylesheet and would restyle parts of the plugin area. Each keeps its
# weight and simply stops at the plugin area's edge.
SITE_ONLY = [
    "figcaption{font-size:13.5px;",
    "td a{display:block;",
    "td a+a{margin-top:9px}",
    "textarea{resize:vertical;",
]


def fence_site_css(site):
    for rule in SITE_ONLY:
        sel, rest = rule.split("{", 1)
        site = once(site, "\n" + rule, "\n" + sel + ":where(:not(.plg *)){" + rest)
    return site


def scope_css(css):
    css = re.sub(r"/\*.*?\*/", "", css, flags=re.S)
    out, dropped = [], 0
    for line in css.split("\n"):
        s = line.strip()
        if not s:
            continue
        if s.startswith("@media"):
            m = re.match(r"(@media[^{]*\{)(.*)$", s)
            out.append(m.group(1))
            s = m.group(2).strip()
            closing = False
            if s.endswith("}}"):          # a whole block on one line
                s, closing = s[:-1], True
            if s:
                for rule in re.findall(r"[^{}]+\{[^{}]*\}", s):
                    r = scope_rule(rule)
                    dropped += r is None
                    out.append(r or "")
            if closing:
                out.append("}")
            continue
        if s == "}":
            out.append("}")
            continue
        assert re.fullmatch(r"[^{}]+\{[^{}]*\}", s), s[:120]
        r = scope_rule(s)
        dropped += r is None
        out.append(r or "")
    return "\n".join(x for x in out if x), dropped


def scope_rule(rule):
    sel, body = rule.split("{", 1)
    sels = [scope_selector(REN_RE.sub(lambda m: ".p-" + m.group(1), x)) for x in split_selectors(sel)]
    sels = [x for x in sels if x]
    return (",".join(sels) + "{" + body) if sels else None

    js = once(js, "function $$(sel, root) { return Array.prototype.slice.call((root || document).querySelectorAll(sel)); }",
              "function $$(sel, root) { return Array.prototype.slice.call((root || PLG).querySelectorAll(sel)); }")
    # one router for every view of the site; only the plugin area's views give up their ids
    js = once(js, "var views = $$('[data-view]');", "var views = $$('[data-view]', document);\n  var mastLinks = $$('.mast .nav a', document);\n  var NAVCUR = { report: 'research', contribute: 'about' };")

              "views.forEach(function (v) { viewByName[v.getAttribute('data-view')] = v; if (PLG.contains(v)) { v.removeAttribute('id'); } });")
    js = once(js, "    navEl.setAttribute('data-group', view.getAttribute('data-group') || 'page');",
              "    var inPlg = PLG.contains(view);\n"

    i = site.index("<h2>Start with the part you need</h2>")
    start = site.rindex('<section class="sec">', 0, i)
    end = site.index("</section>", i) + len("</section>")
    assert site[start:end].count("<section") == 1 and '<ul class="idx">' in site[start:end]
    return site[:start] + section + site[end:]


def merge_llms(text):
    """Add the plugin pages to the site's llms.txt, once, ahead of the part on contributing."""
    if "## Plugins" in text:
        return text
    mark = "## Contributing"
    if mark in text:
        return text.replace(mark, LLMS + "\n" + mark, 1)
    return text.rstrip("\n") + "\n\n" + LLMS


def build(site_path, plug_dir, out_path, preview=False, llms_path=None, llms_out=None, use_cases=None):
    site = read(site_path)
    assert 'id="plg"' not in site, "this page already holds the plugin area: give the page as build.py made it"

    def plug(name):
        # in the kit the plugin page is kept as index.html.txt, so that it is served as text and not as a second page
        p = os.path.join(plug_dir, name)
        return read(p if os.path.exists(p) else p + ".txt")
    mine = plug("index.html")
    # 1. navigation: Plugins is a section of this page
    assert site.count(PLUGINS_URL) == 3
    # the arrow that marks a link leaving the site is written as a character or as an entity, by the build that made the page
    site, n = re.subn(r'<a href="%s">Plugins <span aria-hidden="true">(?:&#8599;|\u2197)</span></a>' % re.escape(PLUGINS_URL), '<a href="#plugins">Plugins</a>', site)
    assert n == 2, n
    site = once(site, f'<li><a href="{PLUGINS_URL}"><b>See the labs\' plugins</b>', '<li><a href="#plugins"><b>See the labs\' plugins</b>')
    site = once(site, "and shorter pages for four more. A preview that opens as a page of its own.</span>", "and shorter pages for four more.</span>")
    assert PLUGINS_URL not in site
    # 5. the home page leads with a finding
    site = new_home(site)
    # 6. the use-case index takes the place of the list of the site's shelves (the staging copy, until the owner says otherwise)
    uc = json.loads(read(use_cases)) if use_cases else None
    if uc:
        site = use_case_index(site, uc)
    # 2. the plugin area
    site = once(site, "</main>", plugin_markup(mine) + "</main>")
    # 3. styles
    site = fence_site_css(site)
    scoped, dropped = scope_css(plug("entry.css"))
    site = once(site, "</style>", "\n/* ---- the plugin area, scoped to .plg ---- */\n" + scoped + EXTRA_CSS + (USE_CASES_CSS if uc else "") + (PREVIEW_CSS if preview else "") + "</style>")
    # 4. scripts: the plugin area's data and script follow the site's, and one router serves every view
    # the site's script still opens the first view itself, so the rest of it starts as it does today; from then on
    # the plugin area's router, which knows every view, answers each change of address
    site = once(site, "addEventListener('hashchange',()=>route(true));\nroute(false);", "route(false);")
    # the report's contents marker looks only at the report, now that the plugin area has contents lists of its own
    site = once(site, "const links=$$('.toc a'),heads=$$('.art>h2[id]'),rep=$('#report');", "const links=$$('#report .toc a'),heads=$$('#report .art>h2[id]'),rep=$('#report');")
    data = re.search(r'<script type="application/json" id="data">(.*?)</script>', mine, re.S).group(1)
    aimd = re.search(r'<script type="application/json" id="ai-md">(.*?)</script>', mine, re.S).group(1)
    tail = ('<script type="application/json" id="plg-data">' + data + '</script>\n<script type="application/json" id="ai-md">' + aimd
            + "</script>\n<script>\n" + patch_app(plug("app.js")) + "</script>\n</body>")
    site = once(site, "</body>", tail)
    if preview:
        site = re.sub(r"<title>.*?</title>", f"<title>{PREVIEW_TITLE}</title>", site, count=1)
        site = once(site, '<header class="mast">', STRIP + '\n<header class="mast">')
    site = once(site, "</head>", '<link rel="alternate" type="application/json" href="skills.json" title="The 36 skills of the Claude Sales plugin as data">\n</head>')
    if uc:
        # every address the index names opens something: an element of the page, or a view
        block = site[site.index('<section class="sec" id="use-cases">'):]
        block = block[:block.index("</section>")]
        for target in sorted(set(re.findall(r'href="#([^"]+)"', block))):
            assert f'id="{target}"' in site or f'data-view="{target}"' in site, "the use-case index links to #%s, which the page does not hold" % target
    os.makedirs(os.path.dirname(os.path.abspath(out_path)), exist_ok=True)
    open(out_path, "w", encoding="utf-8").write(site)
    ids = re.findall(r'\sid="([^"]+)"', site)
    dup = sorted({i for i in ids if ids.count(i) > 1})
    print("built", out_path, len(site.encode()), "bytes;", len(re.findall(r'data-view="', site)), "views; css rules dropped:", dropped, "; duplicate ids:", dup,
          "; preview strip:", "yes" if preview else "no", "; use-case index:", "yes" if uc else "no")
    if llms_path and os.path.exists(llms_path):
        text = merge_llms(read(llms_path))     # read before the file is opened for writing: the two can be one file
        open(llms_out or llms_path, "w", encoding="utf-8").write(text)
        print("llms.txt: plugin pages listed in", llms_out or llms_path)


def main():
    ap = argparse.ArgumentParser(description="Put the plugin area inside the Modern GTM site.")
    ap.add_argument("--site", default=os.path.join(HERE, "dist", "index.html"), help="the site's page as build.py made it")
    ap.add_argument("--plugins", default=os.path.join(HERE, "plugins"), help="folder with the plugin page's index.html, entry.css and app.js")
    ap.add_argument("--out", help="where to write the merged page (default: over --site, keeping a copy of it)")
    ap.add_argument("--llms", help="the site's llms.txt (default: llms.txt beside --site)")
    ap.add_argument("--llms-out", help="where to write the merged llms.txt (default: over --llms)")
    ap.add_argument("--preview", action="store_true", help="mark the page as a staging copy")
    ap.add_argument("--use-cases", help="use_cases.json: put the use-case index on the home page, in place of the list of the site's parts")
    a = ap.parse_args()
    site_path, out = a.site, a.out
    if not out:
        # in place: keep the page as build.py made it, and start from that copy on a second run
        keep = os.path.join(os.path.dirname(os.path.abspath(site_path)), "index.before-merge.html")
        if 'id="plg"' not in read(site_path):
            open(keep, "w", encoding="utf-8").write(read(site_path))
        site_path, out = keep, a.site
    llms = a.llms or os.path.join(os.path.dirname(os.path.abspath(a.site)), "llms.txt")
    # with --out the inputs stay as they are: the merged llms.txt goes beside the merged page
    llms_out = a.llms_out or (os.path.join(os.path.dirname(os.path.abspath(a.out)), "llms.txt") if a.out else None)
    build(site_path, a.plugins, out, a.preview, llms, llms_out, a.use_cases)


if __name__ == "__main__":
    main()
