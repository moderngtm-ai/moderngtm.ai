def prerender(path):
    from playwright.sync_api import sync_playwright
    js = """() => {
      const q = s => [...document.querySelectorAll(s)].map(e => e.innerHTML);
      const out = {chart: q('[data-chart]'), strip: q('[data-strip]'), tbody: document.querySelector('#reg tbody').innerHTML, ids: {}, vals: {}, checks: []};
      for (const i of ['gradeSeg','regCount','uOut','cText','kCount','mLevel','mWhy','mNext','mProf','rubric','ledger']) out.ids[i] = document.getElementById(i).innerHTML;
      for (const e of document.querySelectorAll('#cForm input, #cForm textarea')) { if (e.tagName === 'TEXTAREA') e.textContent = e.value; else e.setAttribute('value', e.value); }
      out.ids.cForm = document.getElementById('cForm').innerHTML;
      for (const e of document.querySelectorAll('#uForm input')) out.vals[e.id] = e.value;
      out.checks = [...document.querySelectorAll('#kList input')].filter(e => e.checked).map(e => e.id);
      return out; }"""
    with sync_playwright() as pw:
        br = pw.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        pg = br.new_page(viewport={'width': 1280, 'height': 900})
        pg.goto('file://' + str(path)); pg.wait_for_timeout(700)
        out = pg.evaluate(js); br.close()
    return out

def inject(html, pre):
    s = BeautifulSoup(html, 'html.parser')
    def fill(el, inner):
        el.clear(); el.append(BeautifulSoup(inner, 'html.parser'))
    charts, strips = s.select('[data-chart]'), s.select('[data-strip]')
    assert len(charts) == len(pre['chart']) and len(strips) == len(pre['strip'])
    for el, inner in zip(charts, pre['chart']): fill(el, inner)
    for el, inner in zip(strips, pre['strip']): fill(el, inner)
    fill(s.select_one('#reg tbody'), pre['tbody'])
    for i, inner in pre['ids'].items(): fill(s.find(id=i), inner)
    for i, v in pre['vals'].items(): s.find(id=i)['value'] = v
    for i in pre['checks']: s.find(id=i)['checked'] = ''
    return str(s)
