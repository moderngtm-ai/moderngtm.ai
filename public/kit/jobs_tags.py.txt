# The job of each record is kept in job_tags_by_key.json, one line for each record, keyed as the register's own files
# key a record: the organization, a bar, then the first 40 characters of the finding. So a record keeps its tag when
# the register grows or is put in another order. Until November 23, 2026 the tags were kept here by record number.
TAGS_FILE = os.path.join(HERE, "job_tags_by_key.json")


def key_of(r):
    return r["organization"] + "|" + r["finding"][:40]


def load_tags(ev):
    """(job, note, status) for each record of the register, in the register's order. Stops on a record with no line
    and on a line whose record is gone, and names them."""
    by_key = json.load(open(TAGS_FILE, encoding="utf-8"))["tags"]
    keys = [key_of(r) for r in ev]
    assert len(set(keys)) == len(keys), "two records share a key"
    missing = [k for k in keys if k not in by_key]
    stale = [k for k in by_key if k not in set(keys)]
    if missing or stale:
        lines = ["job_tags_by_key.json and the register do not match."]
        if missing:
            lines.append(f"{len(missing)} records have no job yet. Add a line for each:")
            lines += ["  " + k for k in missing]
        if stale:
            lines.append(f"{len(stale)} lines name no record. If a finding's first 40 characters changed, change its key:")
            lines += ["  " + k for k in stale]
        sys.exit("\n".join(lines))
    return [(by_key[k]["job"], by_key[k].get("note", ""), by_key[k].get("status", "Accept")) for k in keys]

def en_list(items):
    """'8', '8 and 87', '8, 87 and 88': the site's lists take no comma before the last item."""
    return items[0] if len(items) == 1 else ", ".join(items[:-1]) + " and " + items[-1]
