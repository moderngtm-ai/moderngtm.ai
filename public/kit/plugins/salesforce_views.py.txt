def code(name):
    return f"<code>{e(name)}</code>"


def schip(i):
    return f'<a class="chip" href="#sfk-{i}">{e(i)}</a>'


def cchip(i):
    return f'<a class="chip" href="#skill-{i}">{e(i)}</a>'


def join(items, last=" and "):
    items = list(items)
    return items[0] if len(items) == 1 else ", ".join(items[:-1]) + last + items[-1]


def a(url, text):
    return f'<a href="{e(url)}">{text}</a>'


def slug(text):
    return re.sub(r"[^a-z0-9]+", "-", text.lower()).strip("-")


def catalog():
    out = []
    for group, rows in S.GROUPS:
        group = group[0] + group[1:].lower()   # Salesforce's groups, in the site's sentence case
        body = []
        for i, name, does, writes, job in rows:
            mark = '<span class="t">Can create and update records</span>' if writes else '<span class="nil">No</span>'
            body.append(f'<tr id="sfk-{i}"><th scope="row">{code(i)}<span class="sfn">{e(name)}</span></th>'
                        f'<td class="fmt" data-l="What it does">{e(does)}</td>'
                        f'<td class="fmt" data-l="GTM job, our read">{e(job) if job else "Outside the 13 jobs"}</td>'
                        f'<td class="fmt" data-l="Records">{mark}</td></tr>')
        out.append(f'<div class="sfgrp"><h3 id="sf-g-{slug(group)}">{e(group)} <small>{len(rows)} {"skill" if len(rows) == 1 else "skills"}</small></h3>'
                   f'<div class="scroll" tabindex="0" role="region" aria-label="{e(group)}: skills"><table class="rt dt skt stack wide sft"><thead><tr>'
                   '<th scope="col">Skill</th><th scope="col">What it does</th><th scope="col">GTM job <small>Our read</small></th><th scope="col">Records</th></tr></thead><tbody>'
                   + "".join(body) + "</tbody></table></div></div>")
    return "\n".join(out)

def by_job():

    rows = []

        sf = "".join(schip(s["id"]) for s in S.SKILLS if s["job"] == j) or none

def check(desc_path="/home/claude/sfic/sf_ref_descriptions.json"):
    """Hold the entry to Salesforce's skills reference as read: counts, IDs, the marks and our own wording."""
    assert len(SF_IDS) == len(set(SF_IDS)) == S.SKILLS_N == 37 and len(S.GROUPS) == 8
    assert sorted(SF_WRITES) == sorted(["account-plan", "update-opportunity", "assign-target-to-sdr", "summarize-conversation-transcript", "schedule-meeting", "log-activity"])
    assert all(s["job"] is None or s["job"] in PD.JOB_NAMES for s in S.SKILLS)
    assert [s["id"] for s in S.SKILLS if s["job"] is None] == ["voice-profile"]
    assert (len(SHARED), len(ONLY_SF), len(ONLY_AN)) == (27, 10, 9), (len(SHARED), len(ONLY_SF), len(ONLY_AN))
    assert len(SHARED_SF_WRITES) == 4 and len(SHARED_CLAUDE_WRITES) == 10, (SHARED_SF_WRITES, SHARED_CLAUDE_WRITES)
    claude_job = {i: j for j, ids in D.JOBS for i in ids}
    assert all(S.BY[i]["job"] == claude_job.get(i) for i in SHARED), [i for i in SHARED if S.BY[i]["job"] != claude_job.get(i)]
    assert len(S.COPILOT_IDS) == len(set(S.COPILOT_IDS)) == 38 and len(CO_SHARED) == 35 and set(S.COPILOT_WRITES) <= set(S.COPILOT_IDS)
    for s in S.SKILLS:
        assert s["does"].endswith(".") and s["does"] == s["does"].strip(), s["id"]
    if not os.path.exists(desc_path):
        return f"{S.SKILLS_N} skills held to the reference; wording not checked: no copy of Salesforce's descriptions"
    theirs = json.load(open(desc_path, encoding="utf-8"))["descriptions"]
    assert sorted(theirs) == sorted(SF_IDS)

    def words(t):
        return re.findall(r"[a-z0-9']+", t.lower().replace("’", "'"))

    def longest(x, y):
        x, y, best = words(x), words(y), 0
        for i in range(len(x)):
            for k in range(len(y)):
                n = 0
                while i + n < len(x) and k + n < len(y) and x[i + n] == y[k + n]:
                    n += 1
                best = max(best, n)
        return best
    runs = {i: longest(S.BY[i]["does"], theirs[i]) for i in SF_IDS}
    worst = max(runs.values())
    assert worst <= 4, {i: n for i, n in runs.items() if n > 4}
    return f"{S.SKILLS_N} skills held to the reference; longest run of words shared with Salesforce's descriptions: {worst}"
