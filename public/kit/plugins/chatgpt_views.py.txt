def code(name):
    return f"<code>{e(name)}</code>"


def gchip(name):
    return f'<a class="chip" href="#gpt-{name}">{e(name)}</a>'


def join(items, last=" and "):
    items = list(items)
    return items[0] if len(items) == 1 else ", ".join(items[:-1]) + last + items[-1]


def slots(text):
    return re.sub(r"\[([^\]]+)\]", lambda m: f'<span class="slot">[{m.group(1)}]</span>', e(text))


def link_names(text):
    """Skill names inside running text become links to their cards."""
    out = e(text)
    for name in sorted(G.BY, key=len, reverse=True):
        out = re.sub(r"(?<![\w>#-])" + re.escape(name) + r"(?![\w-])", f'<a class="skl" href="#gpt-{name}"><code>{name}</code></a>', out)
    return out


def flags(s):
    out = [f'<span class="t">{G.FOR[s["who"]]}</span>']
    if s["sflag"] == "yes":
        out.append('<span class="t">Can be scheduled</span>')
    if s["mflag"] == "yes":
        out.append('<span class="t">For a named seller</span>')
    return "".join(out)

def catalog():
    out = []
    i = 0
    for key, title, sub in G.GROUPS:
        rows = [s for s in G.SKILLS if s["group"] == key]
        out.append(f'<div class="ggrp"><h3 id="gpt-g-{key}">{e(title)} <small>{e(sub)}</small></h3>')
        for s in rows:
            out.append(skill(s, G.SKILLS.index(s)))
            i += 1
        out.append("</div>")
    assert i == G.SKILLS_N
    return "\n".join(out)

def check(reply_path="/home/claude/chatgpt/skills_reported.json"):
    """Hold the entry to the reply it restates, and keep the plugin's own wording out of it."""
    import json
    import os
    names = [s["name"] for s in G.SKILLS]
    assert len(names) == len(set(names)) == G.SKILLS_N == 18
    assert len(SCHED) == 7 and len(MGR) == 5, (len(SCHED), len(MGR))
    assert sum(n for _c, n, _x in G.WORKS_WITH) == G.INTEGRATIONS + 1 and len(G.WORKS_WITH) == 10
    assert sum(len(x) for _c, _n, x in G.WORKS_WITH) + G.OPAQUE == G.INTEGRATIONS
    assert sum(len(x) for _c, _n, x in G.WORKS_WITH) - 2 + len(G.README_EXTRA) == G.README_NAMES
    assert len(G.PAGE_APPS) + G.PAGE_APPS_MORE == 41
    placed = [x for _j, ids in G.JOBS for x in ids]
    assert sorted(placed + G.OUTSIDE_JOBS) == sorted(names)
    for s in G.SKILLS:
        for field in ("does", "when", "ask", "needs", "back", "reads", "makes", "asks", "scope", "sched", "noapps", "company", "wont", "next"):
            assert s[field] and s[field].strip() == s[field], (s["name"], field)
        assert s["ask"].startswith("@Sales ")
        assert (s["sflag"] != "") or s["sched"] == G.NS, s["name"]
    if not os.path.exists(reply_path):
        return "not checked against the reply: no copy here"
    rep = {r["name"]: r for r in json.load(open(reply_path, encoding="utf-8"))}
    assert sorted(rep) == sorted(names)
    words = lambda t: re.findall(r"[a-z0-9']+", t.lower().replace("’", "'"))

    def grams(t, n):
        w = words(t)
        return {" ".join(w[i:i + n]) for i in range(len(w) - n + 1)}
    worst_plugin = worst_relay = 0

    def longest(a, b, top):
        return max((n for n in range(2, top + 1) if grams(a, n) & grams(b, n)), default=0)
    for s in G.SKILLS:
        r = rep[s["name"]]
        ours = " . ".join(s[f] for f in ("title", "does", "when", "ask", "needs", "back", "reads", "makes", "asks", "scope", "sched", "noapps", "company", "wont", "next"))
        # the plugin's own text: the description, the example prompt and every phrase the reply put in quotation marks
        theirs = r["description"] + " . " + r["invoke"] + " . " + " . ".join(re.findall(r"\u201c([^\u201d]+)\u201d", " ".join(v for v in r.values() if isinstance(v, str))))
        a = longest(ours, theirs, 6)
        assert a < 5, (s["name"], "shares a run of", a, "words with the plugin's text", sorted(grams(ours, a) & grams(theirs, a))[:2])
        # ChatGPT's own summary is not the plugin's text, but the entry should not lean on its wording either
        relay = " . ".join(r[f] for f in ("does", "use_when", "inputs_required", "inputs_optional", "reads", "writes", "approval", "output", "hands_to", "follows", "scope", "schedule", "without_apps", "company_settings", "limits"))
        b = longest(ours, relay, 9)
        assert b < 8, (s["name"], "shares a run of", b, "words with the reply", sorted(grams(ours, b) & grams(relay, b))[:2])
        worst_plugin, worst_relay = max(worst_plugin, a), max(worst_relay, b)
    # the whole page, not only the cards
    full_path = os.path.join(os.path.dirname(reply_path), "reply-2026-11-22.txt")
    page = ""
    if os.path.exists(full_path):
        t = open(full_path, encoding="utf-8").read()
        a, b = t.index("\n[\n") + 1, t.index("\n]\n") + 2
        text = html.unescape(re.sub(r"<[^>]+>", " ", view())).replace(G.START_PROMPT, " ")
        quoted = " . ".join(re.findall(r"\u201c([^\u201d]+)\u201d", t) + [r["description"] + " . " + r["invoke"] for r in rep.values()])
        pa, pb = longest(text, quoted, 8), longest(text, t[:a] + " . " + t[b:], 10)
        assert pa < 6, ("the page shares a run of", pa, "words with the plugin's text", sorted(grams(text, pa) & grams(quoted, pa))[:3])
        assert pb < 8, ("the page shares a run of", pb, "words with the reply's prose", sorted(grams(text, pb) & grams(t[:a] + " . " + t[b:], pb))[:3])
        page = f"; whole page: {pa} and {pb}"
    return (f"18 skills held to the reply; longest run of words shared with the plugin's own text: {worst_plugin}; "
            f"with ChatGPT's wording: {worst_relay}{page}")
