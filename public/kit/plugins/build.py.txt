def skl(sid):
    """A skill named in running text: code style, linked to its row."""
    assert sid in SK, sid
    return f'<a class="skl" href="#skill-{sid}"><code>{sid}</code></a>'


def tags(items, cls=""):
    return "".join(f'<span class="t{cls}">{e(i)}</span>' for i in items)

def rows_html(pairs):
    return '<div class="rows">' + "".join(f'<div><span class="l">{a}</span><span>{b}</span></div>' for a, b in pairs) + "</div>"


def lean_rows(s):
    parts = C.lean_parts(s)
    pairs = []
    if s["need"]:
        pairs.append(("Needs", e(s["need"])))
    for label, keys in parts.items():
        if label == "Reads through" and not keys:
            pairs.append((label, "No tool of its own."))
            continue
        if not keys:
            continue
        body = '<span class="chips">' + tags([C.tagform(k, s) for k in keys], " need" if label in ("Built around", "Needs, or asks once") else "")
        if label == "Reads through":
            body += '<span class="note">or an export or a paste</span>'
        pairs.append((label, body + "</span>"))
    if s["leannote"]:
        pairs.append(("Note", e(s["leannote"])))
    return rows_html(pairs)


def value_rows(s):
    if not s["seg"]:
        return f'<p style="margin:0">{e(s["why"])}</p>'
    pairs = []
    for v in (3, 2, 1, 0):
        ms = [(k, n) for k, n, *_ in D.MOTIONS if s["seg"][k][0] == v]
        if ms:
            body = "".join(f'<span class="t{" need" if v == 3 else ""}{" off" if v == 0 else ""}" data-m="{k}">{e(n)}</span>' for k, n in ms)
            pairs.append((D.VALUE[v], f'<span class="chips">{body}</span>'))
    often = C.how_often(s)
    if often:
        pairs.append(("How often", e(often)))
    return rows_html(pairs) + f'<p style="margin:10px 0 0">{e(s["why"])}</p>'


def slots(text):
    """A request, with the parts the reader fills in marked."""
    return re.sub(r"\[([^\]]+)\]", lambda m: f'<span class="slot">[{m.group(1)}]</span>', e(text))


def low(label):
    """A label set in running text: lower case, unless its first word is an acronym or a coined name."""
    first = label.split()[0]
    return label if (first.isupper() or any(c.isupper() for c in first[1:])) else label[0].lower() + label[1:]


def seg(sid, name, label, options):
    """A two-way choice inside a run card. The first option starts chosen."""
    btns = "".join(f'<button type="button" data-set="{name}" data-v="{v}" aria-pressed="{"true" if i == 0 else "false"}">{e(word)}</button>'
                   for i, (v, word) in enumerate(options))
    return f'<span class="vg" role="group" aria-label="{label}"><span class="vl" aria-hidden="true">{label}</span><span class="seg">{btns}</span></span>'


def model_of(s):
    """(model, reason, extra) for the model a skill starts on. Our read."""
    tier = C.tier_of(s)
    model, why = D.MODEL_TIERS[tier]
    return tier, model, why


def run_card(s):
    """How to use a skill on an ordinary day: when, what to say, what to give it and what comes back."""
    sid = s["id"]
    r = D.RUN[sid]
    f = C.FACTS[sid]
    needed = [low(C.TOOL_LABEL[t["type"]]) for t in f["tools"] if t["required"]]
    conn = f"With {join(needed)} connected, the request is enough to start." if needed else "With your tools connected, the request is enough to start."
    if r["asks"]:
        conn += " " + r["asks"]
    conn += " It says what it could not see."
    ctl = [seg(sid, "src", "Your tools", [("conn", "Connected"), ("files", "Working from files")])]
    alts = []
    if r["team"]:
        first, second, lead = r["words"]
        ctl.append(seg(sid, "scope", "Scope", [("me", first), ("team", second)]))
        alts.append(f'<p class="alt nojs"><b>{e(lead)}</b> <span class="sayt">{slots(r["team"])}</span></p>')
    if r["sched"]:
        ctl.append(seg(sid, "time", "Timing", [("now", "On request"), ("sched", "On a schedule")]))
        alts.append(f'<p class="alt nojs"><b>On a schedule.</b> <span class="sayt">{slots(r["sched"])}</span></p>')
    data = "".join(f' data-{k}="{e(r[k], quote=True)}"' for k in ("team", "sched", "both") if r[k])

    also = ('<p class="alt also">Also answers to ' + " or ".join(f"\u201C{e(a)}\u201D" for a in others) + ".</p>") if others else ""
    return f'''<div class="run" data-run="{sid}">

def act_rows(s):
    """The four facts on what a skill can do. A fact the file does not give is said in plain words."""
    pairs = []
    for k, label in ACT_LABELS:
        text = s["run"] if k == "run" else s[k][1]
        yes = bool(s["run"]) if k == "run" else s[k][0] != "no"
        pairs.append((label, f'<span class="{"yes" if yes else "nope"}">{e(text or ACT_NONE[k])}</span>'))
    return '<div class="rows acts4">' + "".join(f'<div><span class="l">{a}</span><span>{b}</span></div>' for a, b in pairs) + "</div>"


def tool_tags(s, f):
    """A file's tools table, split by direction: what the skill reads, and where it can deliver something."""
    reads, sends = [], []
    for t in f["tools"]:
        way, what = s["io"].get(t["type"], ("r", ""))
        label = C.TOOL_LABEL[t["type"]]
        if way in ("r", "rd"):
            reads.append(f'<span class="t{" need" if t["required"] else ""}">{e(label)}{", needed" if t["required"] else ""}</span>')
        if way in ("d", "rd"):
            sends.append(f'<span class="t">{e(label)}<span class="tw">: {e(what)}</span></span>')
    if s["rnote"]:
        reads.append(f'<span class="cond rn">{e(s["rnote"])}</span>')
    return "".join(reads), "".join(sends)

def routines():
    out = ['<div class="routines" id="routines">']

def matrix(mk):
    """The grid, written quadrant by quadrant so a phone reads it in order of use; the page places it as a grid."""
    out = ['<div class="mx" id="mx" role="group" aria-label="Skills placed by value and by frequency">',
           '<div class="corner" style="--r:1;--c:1"></div>']
    for f in (1, 2, 3, 4):
        out.append(f'<div class="ch" style="--r:1;--c:{f + 1}">{D.FREQ[f]}</div>')
    rowof = {3: 3, 2: 4, 1: 6, 0: 7}
    for v in (3, 2, 1, 0):
        out.append(f'<div class="rh" style="--r:{rowof[v]};--c:1">{D.VALUE[v]}<small>{e(D.VALUE_NOTE[v])}</small></div>')
    out.append('<div class="fill" style="--r:2;--c:1"></div><div class="fill b" style="--r:5;--c:1"></div>')
    order = [("ho", "hi-often", 2, "4 / span 2", [(3, 3), (3, 4), (2, 3), (2, 4)]),
             ("hr", "hi-rare", 2, "2 / span 2", [(3, 1), (3, 2), (2, 1), (2, 2)]),
             ("lo", "lo-often", 5, "4 / span 2", [(1, 3), (1, 4)]),
             ("lr", "lo-rare", 5, "2 / span 2", [(1, 1), (1, 2), (0, 1)])]   # a poor fit has no frequency: one cell, the width of the grid
    for q, qk, qrow, qcol, cells in order:
        name, desc = D.QUADS[qk]
        cell_html = []
        total = 0
        for v, f in cells:
            ids = [s["id"] for s in D.SKILLS if s["seg"] and s["seg"][mk] == [v, f]]
            total += len(ids)
            cell_html.append(f'<div class="cell q-{q}{"" if ids else " e"}" data-v="{v}" data-f="{f}" style="--r:{rowof[v]};--c:{"2 / span 4" if v == 0 else f + 1}">'
                             f'<span class="lab">{D.VALUE[v] if v == 0 else D.VALUE[v] + ", " + D.FREQ[f].lower()}</span>{chips(ids)}</div>')
        out.append(f'<div class="ql q-{q}" data-q="{q}" style="--r:{qrow};--c:{qcol}"><b>{name}</b><small>{desc}</small>'
                   f'<span class="none"{"" if total == 0 else " hidden"}>None in this motion.</span></div>')
        out.extend(cell_html)
    out.append("</div>")
    return "".join(out)

def shape(c):
    """What one run returned, from the files it left."""
    if not c["extra_files"]:
        return f'All in the reply, {c["words"]:,} words.'
    kind = "page" if c["extra_files"][0].endswith((".html", ".htm")) else "document"
    return f'A reply of {c["words"]:,} words and a {kind} of {c["file_words"]:,}.'


def pilot_summary():
    """One row for each model, summed over the six skills."""
    P = PILOT
    out = ['<div class="scroll" tabindex="0" role="region" aria-label="The test in sum, by model"><table class="rt pt sum stack wide"><thead><tr><th scope="col">Model</th>'
           '<th scope="col">Planted items found</th><th scope="col">Numbers in the key right</th><th scope="col">Guardrails kept</th><th scope="col">Slips the markers noted</th>'
           '<th scope="col">Time for one run</th><th scope="col">What came back</th></tr></thead><tbody>']
    for m in P["models"]:
        cs = [P["cells"][sk][m] for sk in P["skills"]]
        assert all(c["done"] and "items" in c for c in cs), m
        assert P["model_ids"][m] == MODEL_ID[m], (m, P["model_ids"][m])
        it, of = sum(c["items"]["score"] for c in cs), sum(c["items"]["of"] for c in cs)
        nu, nof = sum(c["numbers"]["score"] for c in cs), sum(c["numbers"]["of"] for c in cs)
        gk, gof = sum(c["guardrails"]["kept"] for c in cs), sum(c["guardrails"]["of"] for c in cs)
        ts = sorted(c["seconds"] for c in cs)
        files = sum(1 for c in cs if c["extra_files"])
        totals = sorted(c["words"] + c["file_words"] for c in cs)
        if files == len(cs):
            back = "A short reply with a separate page or document, every time."
        elif files == 0:
            back = "Everything in one reply, every time."
        else:
            back = (f"A short reply with a separate page or document in {num(files)} of {num(len(cs))} runs, "
                    f"and everything in one reply in the other {num(len(cs) - files)}.")
        back += f" {totals[0]:,} to {totals[-1]:,} words in all."
        out.append(f'<tr><th scope="row">{MODEL_NAME[m]}</th><td data-l="Items found">{frac(it, of)}</td><td data-l="Numbers in the key">{frac(nu, nof)}</td><td data-l="Guardrails kept">{gk} of {gof}</td>'
                   f'<td class="fmt" data-l="Slips">{e(D.PILOT_SLIPS[m])}</td><td data-l="Time for a run">{secs(ts[0])} to {secs(ts[-1])}</td><td class="fmt" data-l="What came back">{back}</td></tr>')
    out.append("</tbody></table></div>")
    return "".join(out)
