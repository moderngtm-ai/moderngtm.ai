def code(name):
    return f"<code>{e(name)}</code>"


def plugin_rows(lab):
    out = ['<ul class="plist">']
    for addr, name, by, facts, what, depth in PAGES:
        if by != lab:
            continue
        out.append(f'<li><a class="pl" href="#{addr}"><span class="pn">{e(name)}</span><span class="pf">{e(facts)}</span>'
                   f'<span class="pd">{e(what)}</span><span class="pgo">{e(depth)}</span></a></li>')
    out.append("</ul>")
    return "".join(out)


def skill_table(rows, label, who=False):
    head = '<th scope="col">Skill</th><th scope="col">What it does</th>' + ('<th scope="col">Who may reach for it</th>' if who else "")
    out = [f'<div class="scroll" tabindex="0" role="region" aria-label="{label}"><table class="rt dt skt stack wide"><thead><tr>{head}</tr></thead><tbody>']
    for r in rows:
        extra = f'<td class="fmt" data-l="Who, in our read">{e(r[2])}</td>' if who else ""
        out.append(f'<tr><th scope="row">{e(r[0])}</th><td class="fmt" data-l="What it does">{e(r[1])}</td>{extra}</tr>')
    out.append("</tbody></table></div>")
    return "".join(out)


def leader_skills():
    out = ['<div class="lsk">']
    for key, rows in P.ADJACENT.items():
        out.append(f'<details class="more"><summary>{e(NAME[key])} <span class="cnt">{len(rows)} of {COUNT[key]} skills, version {VER[key]}</span></summary>'
                   f'{skill_table(rows, NAME[key] + " skills a revenue team may use", who=True)}</details>')
    out.append("</div>")
    return "".join(out)

def check(idx_path="/home/claude/plugins_index.json"):
    """Hold these pages to the repository: versions, counts and every skill name."""
    import json
    import os
    if not os.path.exists(idx_path):
        return "not checked: no copy of the repository index"
    idx = json.load(open(idx_path, encoding="utf-8"))
    for key, _name, ver, n, _level, _note in P.ANTHROPIC:
        assert idx[key]["version"] == ver and len(idx[key]["skills"]) == n, key
    for plug, rows in list(P.LISTED.items()) + [(k, [(a, b) for a, b, _c in v]) for k, v in P.ADJACENT.items()]:
        names = {n for n, _ in idx[plug]["skills"]}
        for sk, _ in rows:
            assert sk in names, (plug, sk)
    sb = {n for n, _ in idx["small-business"]["skills"]}
    assert {s for s, _ in P.LISTED["small-business"]} | set(P.SMALL_BUSINESS_REST) == sb
    return "checked against the repository index; ChatGPT for Sales: " + GV.check() + "; Salesforce in Claude: " + SV.check()
