def _sec(g, sid):
    m = re.search(r'<section id="%s">(.*?)</section>' % sid, g, re.S)
    if not m:
        raise SystemExit("guide: section %s not found" % sid)
    return m.group(1)


def _between(h, start, end):
    i = h.index(start)
    j = h.index(end, i)
    return h[:i], h[i:j], h[j:]


def _skill_links(h):
    """A skill named in running text links to its row. The name keeps its code style."""
    def rep(m):
        sid = m.group(1)
        return f'<a class="skl" href="#skill-{sid}"><code>{sid}</code></a>' if sid in C.SK else m.group(0)
    return re.sub(r"<code>([a-z][a-z-]+)</code>", rep, h)


def _skill_chips(inner):
    """Inside a list of skills, names become the same marked chips used everywhere else on the page."""
    def rep(m):
        sid = m.group(1)
        return C.chip(sid) if sid in C.SK else m.group(0)
    return re.sub(r"<code>([a-z][a-z-]+)</code>", rep, inner)


def _common(h):
    h = re.sub(r' style="[^"]*"', "", h)
    h = h.replace('<th class="caps" scope="col">', '<th scope="col">')
    h = h.replace('<dt class="caps">', "<dt>")
    h = re.sub(r'<span class="caps">(.*?)</span>', r"<h4>\1</h4>", h)
    h = h.replace('class="fine"', 'class="note"')
    h = h.replace('<ul class="list">', "<ul>")
    h = re.sub(r'<ol class="steps">(.*?)</ol>',
               lambda m: '<ol class="wk">' + re.sub(r"<li>(.*?)</li>", r"<li><span>\1</span></li>", m.group(1), flags=re.S) + "</ol>",
               h, flags=re.S)
    h = h.replace('<div class="plain">', '<div class="box">')
    h = h.replace('<div class="duo">', '<div class="g2">').replace('<div class="card">', '<div class="def">')
    return _skill_links(h)


def _table(h, name, wide=False):
    """Give a table the site's stacking pattern: each cell carries its column's label for narrow screens."""
    m = re.search(r'<table class="%s">(.*?)</table>' % name, h, re.S)
    if not m:
        raise SystemExit("guide: table %s not found" % name)
    body = m.group(1)
    labels = [re.sub(r"<[^>]+>", "", x).strip() for x in re.findall(r"<th[^>]*>(.*?)</th>", body, re.S)]

    def row(rm):
        cells = re.findall(r"<td>(.*?)</td>", rm.group(1), re.S)
        if len(cells) != len(labels):
            raise SystemExit("guide: table %s has a row with %d cells" % (name, len(cells)))
        return "<tr>" + "".join(f'<td data-l="{html.escape(l, quote=True)}">{c}</td>' for l, c in zip(labels, cells)) + "</tr>"
    head, tail = body.split("<tbody>", 1)
    tail = re.sub(r"<tr>(.*?)</tr>", row, tail, flags=re.S)
    cls = "stack wide" if wide else "stack"
    return h[:m.start()] + f'<table class="{cls} t-{name}">' + head + "<tbody>" + tail + "</table>" + h[m.end():]


def port(path):
    g = open(path, encoding="utf-8").read()
    out = {}
    # the AI-ready edition is a separate text and keeps every word; the edits apply to the page's own sections
    i = g.index('<script type="application/json" id="ai-md">')
    j = g.index("</script>", i)
    head, edition, tail = g[:i], g[i:j], g[j:]
    for old, new in STYLE_EDITS:
        if (head + tail).count(old) != 1:
            raise SystemExit("guide: expected one match for a style edit, found %d: %s" % ((head + tail).count(old), old))
        head, tail = head.replace(old, new), tail.replace(old, new)
    import json as _json
    esc = lambda t: _json.dumps(t, ensure_ascii=False)[1:-1]          # the edition is one JSON string
    for finding, layer, old, new in CHECK_EDITS:
        if layer == "v":
            if (head + tail).count(old) != 1:
                raise SystemExit("guide: expected one match in the visible text for %s, found %d: %s" % (finding, (head + tail).count(old), old[:60]))
            head, tail = head.replace(old, new), tail.replace(old, new)
        else:
            if edition.count(esc(old)) != 1:
                raise SystemExit("guide: expected one match in the edition for %s, found %d: %s" % (finding, edition.count(esc(old)), old[:60]))
            edition = edition.replace(esc(old), esc(new))
    g = head + edition + tail
    out["check_edits"] = len(CHECK_EDITS)
    out["check_findings"] = CHECK_FINDINGS

    # ------------------------------------------------------------------ opener
    out["deck"] = re.search(r'<p class="deck">(.*?)</p>', g, re.S).group(1).strip()
    out["film"] = re.search(r'(<figure class="film" id="film">.*?</figure>)', g, re.S).group(1)
    out["byline"] = [re.sub(r"<[^>]+>", "", x).strip() for x in
                     re.findall(r"<span>(.*?)</span>", re.search(r'<div class="byline">(.*?)</div>', g, re.S).group(1), re.S)]

    # ------------------------------------------------------------------ what you get
    o = _sec(g, "outcome")
    pre, ledger, post = _between(o, '<div class="ledger"', '<p class="fine">')
    rows = re.findall(r'<div class="ask" role="cell">(.*?)</div><div class="stock" role="cell">(.*?)</div><div class="conf" role="cell">(.*?)</div>', ledger, re.S)
    if len(rows) != 5:
        raise SystemExit("guide: expected five comparison rows, found %d" % len(rows))
    tbl = ['<div class="scroll"><table class="stack t-cmp"><thead><tr><th scope="col">You ask</th><th scope="col">Stock plugin</th><th scope="col">Configured plugin</th></tr></thead><tbody>']
    for a, s, c in rows:
        tbl.append(f'<tr><td data-l="You ask">{a}</td><td data-l="Stock plugin">{s}</td><td data-l="Configured plugin">{c}</td></tr>')
    tbl.append("</tbody></table></div>\n")
    o = pre + "".join(tbl) + post
    pre, cards, post = _between(o, '<div class="cards">', "<h3>What it does not do</h3>")
    items = re.findall(r'<div class="card"><h4><svg.*?</svg>(.*?)</h4><p>(.*?)</p></div>', cards, re.S)
    if len(items) != 4:
        raise SystemExit("guide: expected four benefit blocks, found %d" % len(items))
    o = pre + '<ul class="princ">' + "".join(f"<li><b>{t.strip()}</b>{p.strip()}</li>" for t, p in items) + "</ul>\n" + post
    out["outcome"] = _common(o)

    # ------------------------------------------------------------------ what each role gets
    r = _sec(g, "roles")
    pre, cards, post = _between(r, '<div class="roles">', '<p class="fine"')
    arts = re.findall(r'<article class="role">(.*?)</article>', cards, re.S)
    if len(arts) != len(ROLE_KEYS):
        raise SystemExit("guide: expected six role blocks, found %d" % len(arts))
    blocks = []
    out["role_names"] = {}
    for key, art in zip(ROLE_KEYS, arts):
        name = re.search(r"<h3>(.*?)</h3>", art, re.S).group(1).strip()
        gets = re.search(r'<p class="gets">(.*?)</p>', art, re.S).group(1).strip()
        out["role_names"][key] = name
        rows = []
        for cls, dt, ddcls, dd in re.findall(r'<div(?: class="(written)")?>\s*<dt class="caps">(.*?)</dt>\s*<dd(?: class="(\w+)")?>(.*?)</dd>\s*</div>', art, re.S):
            dd = dd.strip()
            if ddcls == "skills":
                rows.append(f'<div><dt>{dt}</dt><dd class="chips">{_skill_chips(dd)}</dd></div>')
            elif cls == "written":
                rows.append(f'<div><dt>{dt}</dt><dd class="eg">{dd}</dd></div>')
            else:
                rows.append(f"<div><dt>{dt}</dt><dd>{dd}</dd></div>")
        blocks.append(f'<div class="def rolecard" data-grole="{key}"><h3>{name}</h3><p>{gets}</p><dl class="kv">{"".join(rows)}</dl></div>')
    out["role_setup"] = _common(pre + '<div class="g2 rolecards">' + "".join(blocks) + "</div>\n" + post)

    # ------------------------------------------------------------------ who changes what
    s = _sec(g, "seats")
    s = _table(s, "overlap", wide=True)
    out["seats"] = _common(s)

    # ------------------------------------------------------------------ what you configure
    l = _sec(g, "layers")
    l = (l.replace('<div class="levels">', '<div class="lvls">').replace('<div class="level hot">', '<div class="lv hot">')
          .replace('<div class="level">', '<div class="lv">').replace('<div class="level-head">', '<div class="lv-head">'))
    l = l.replace('<span class="tag">the part you customize</span>', '<span class="hl">the part you customize</span>')
    out["layers"] = _common(l)

    # ------------------------------------------------------------------ how to configure it
    t = _sec(g, "steps")
    t = (t.replace('<div class="phases">', '<div class="cphases">').replace('<div class="phase">', '<div class="cphase">')
          .replace('<div class="code">', '<div class="codeblk">').replace('<button class="copy"', '<button class="btn sm"')
          .replace('<div class="quick">', '<div class="box quick">'))
    out["steps"] = _common(t)

    # ------------------------------------------------------------------ the facts to write down
    f = _sec(g, "facts")
    pre, _filt, post = _between(f, '<div class="filter"', '<div class="status">')
    f = pre + "{ROLEPICK}\n" + post
    n = [0]

    def fact(m):
        n[0] += 1
        return f'<div class="fact" id="fact-{n[0]:02d}" data-roles="{m.group(1)}">'
    f = re.sub(r'<div class="fact" data-roles="([a-z ]*)">', fact, f)
    if n[0] != 17:
        raise SystemExit("guide: expected 17 facts, found %d" % n[0])
    f = re.sub(r'<div class="skills">(.*?)</div>', lambda m: '<div class="chips">' + _skill_chips(m.group(1)) + "</div>", f, flags=re.S)
    old = "The role buttons follow the starting map in the role section above."
    if old not in f:
        raise SystemExit("guide: the note under the facts changed")
    f = f.replace(old, "The role choice follows the starting map in the role section above.")
    out["facts"] = _common(f)
    out["fact_ids"] = {html.unescape(name): f"fact-{i:02d}" for i, name in
                       enumerate(re.findall(r'<label for="f\d\d">(.*?)</label>', f), 1)}

    # ------------------------------------------------------------------ do it with Claude
    h = _sec(g, "for-claude")
    h = h.replace('<div class="handoff">', '<div class="hand">').replace('<div class="meta">', '<div class="tbar">')
    out["for_claude"] = _common(h)
    out["ai_md"] = re.search(r'<script type="application/json" id="ai-md">(.*?)</script>', g, re.S).group(1)

    # ------------------------------------------------------------------ tips, pitfalls, example, questions
    p = _sec(g, "tips")
    p = p.replace('<div class="tipflow">', '<div class="tipflow">').replace('<ul class="tips">', '<ul class="princ">')
    out["tips"] = _common(p)
    out["pitfalls"] = _common(_table(_sec(g, "pitfalls"), "fix"))
    out["example"] = _common(_table(_sec(g, "example"), "delta"))
    out["questions"] = _common(_sec(g, "questions"))

    # ------------------------------------------------------------------ sources (they join this entry's list)
    src = _sec(g, "sources")
    out["sources_note"] = re.search(r"<p>(.*?)</p>", src, re.S).group(1).strip()
    out["sources"] = [x.strip() for x in re.findall(r"<li>(.*?)</li>", src, re.S)]
    out["colophon"] = [re.sub(r"<[^>]+>", "", x).strip() for x in
                       re.findall(r"<p>(.*?)</p>", g[g.index("<footer"):g.index("</footer>")], re.S)]
    return out
