# -*- coding: utf-8 -*-
"""Make a copy of the built page for local checks: the two typefaces come from local files in place of
Google Fonts, which the check machine cannot reach. The published page is out/index.html, unchanged."""
import os, re, shutil, sys
HERE = os.path.dirname(os.path.abspath(__file__))
src = open(os.path.join(HERE, "out", "index.html"), encoding="utf-8").read()
faces = '''<style>
@font-face{font-family:"Archivo";src:url(fonts/archivo-latin-wdth-normal.woff2) format("woff2");font-weight:100 900;font-stretch:62% 125%;font-style:normal}
@font-face{font-family:"Source Serif 4";src:url(fonts/source-serif-4-latin-opsz-normal.woff2) format("woff2");font-weight:200 900;font-style:normal}
@font-face{font-family:"Source Serif 4";src:url(fonts/source-serif-4-latin-opsz-italic.woff2) format("woff2");font-weight:200 900;font-style:italic}
</style>'''
out, n = re.subn(r'<link rel="preconnect" href="https://fonts.googleapis.com">.*?display=swap">', lambda m: faces, src, flags=re.S)
assert n == 1
open(os.path.join(HERE, "test", "index.html"), "w", encoding="utf-8").write(out)
# the same page with every disclosure open, for checks that need to see inside them
open(os.path.join(HERE, "test", "index-open.html"), "w", encoding="utf-8").write(re.sub(r"<details(?![^>]* open)", "<details open", out))
shutil.copy(os.path.join(HERE, "out", "skills.json"), os.path.join(HERE, "test", "skills.json"))
print("test copy written")
