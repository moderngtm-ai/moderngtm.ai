def join(items):
    """a, b and c. A serial comma only when an item itself contains 'and'."""
    items = list(items)
    if len(items) < 2:
        return "".join(items)
    if any(" and " in i or " or " in i for i in items) and len(items) > 2:
        return ", ".join(items[:-1]) + ", and " + items[-1]
    if any(" and " in i for i in items) and len(items) == 2:
        return items[0] + ", and " + items[1]
    return ", ".join(items[:-1]) + " and " + items[-1]


def join_or(items):
    items = list(items)
    if len(items) < 2:
        return "".join(items)
    return ", ".join(items[:-1]) + " or " + items[-1]

def check():
    errs = []
    if set(SK) != set(FACTS):
        errs.append("skill ids differ from the plugin: %s" % (set(SK) ^ set(FACTS)))
    seats = {k for k, *_ in D.SEATS}
    for sid, s in SK.items():
        f = FACTS[sid]
        types = [t["type"] for t in f["tools"]]
        need = [t["type"] for t in f["tools"] if t["required"]]
        refs = set(f["refs"]) | {b for a, b in ALIAS if a == sid}
        rec, cust, out = s["rec"], s["cust"], s["out"]
        # the four facts against the file's own write line
        if rec[0] == "own" and f["gated_none"]:
            errs.append(f"{sid}: said to change records, but its write line says none")
        if f["gated_none"] and rec[0] == "own":
            errs.append(f"{sid}: its write line says none")
        if not f["gated_none"] and rec[0] != "own":
            if sid not in SHARE_ONLY:
                errs.append(f"{sid}: its write line lists a write that the entry does not")
            elif not (out[0] == "push" or cust[0] == "send" or sid == "setup"):
                errs.append(f"{sid}: its write line lists a share or a send that the entry does not")
        for part in (rec, cust):
            if part[0] == "no" and part[1] and not part[1].startswith("None"):
                errs.append(f"{sid}: a fact marked no has text that reads as a yes: {part[1][:40]}")
        if out[0] == "no" and out[1] and "is shown in Claude, and its file sets up no send or post for it." not in out[1]:
            errs.append(f"{sid}: its own output is marked as going nowhere, and the text does not say that the file sets up no send or post")
        if "stays in Claude" in " ".join((rec[1], cust[1], out[1], s["run"])):
            errs.append(f"{sid}: a row still says that something stays in Claude, which reads as a limit")
        # directions: every key is a tool in the file's table, and what is needed is read
        for k, (way, what) in s["io"].items():
            if k not in types:
                errs.append(f"{sid}: direction given for {k}, which is not in its tools table")
            if way not in ("d", "rd") or not what:
                errs.append(f"{sid}: bad direction for {k}")
            if k in need and way == "d":
                errs.append(f"{sid}: {k} is needed, so it is read")
        sends_by = {k for k, (way, _w) in s["io"].items()}
        if out[0] == "push" and not (sends_by & {"chat", "email"}):
            errs.append(f"{sid}: shares its output, with no chat or email delivery in its tools")
        if cust[0] == "send" and "email" not in sends_by:
            errs.append(f"{sid}: sends customer email, with no email delivery in its tools")
        if (sid in DOC_OUT) != ("docs" in sends_by):
            errs.append(f"{sid}: documents are a destination only where the tools table gives them as an output")
        if s["cls"] == "W" and not (sends_by & {"crm", "calendar", "sales engagement"}):
            errs.append(f"{sid}: changes records, with no record system among the places it delivers to")
        if s["cls"] == "P" and ("email" in sends_by) != (sid == "daily-briefing"):
            errs.append(f"{sid}: of the skills that share their output, only daily-briefing emails it")
        # other skills, by kind
        for h in s["hand"]:
            if h not in refs:
                errs.append(f"{sid}: hands to {h}, which its file does not name")
        for h in s["draws"]:
            if h not in refs:
                errs.append(f"{sid}: draws on {h}, which its file does not name")
        if set(s["hand"]) & set(s["draws"]):
            errs.append(f"{sid}: a skill is both handed work and drawn on")
        if not set(s["when"]) <= set(s["hand"]) | {"*"}:
            errs.append(f"{sid}: a condition on a handoff it does not have")
        if not set(s["dwhen"]) <= set(s["draws"]):
            errs.append(f"{sid}: a condition on a skill it does not draw on")
        for k in s["quiet"]:
            if k not in s["adds"] or SWG[k] != "decisions" or k in s["must"]:
                errs.append(f"{sid}: a quiet decision must be a decision it uses and never needs")
        for k in s["crmset"]:
            if k not in s["adds"] or SWG[k] != "decisions" or k in s["must"] or k in s["quiet"]:
                errs.append(f"{sid}: a decision taken from the CRM's setup must be a decision it uses, and no other kind")
        for k in s["tagas"]:
            if k not in s["adds"]:
                errs.append(f"{sid}: words given for a switch it does not lean on")
        for a, b in ALIAS:
            if a == sid and not f["names_hygiene_flow"]:
                errs.append(f"{sid}: its file no longer names the crm hygiene flow")
        # scheduled runs
        if f["sched_own"] != bool(s["run"]):
            errs.append(f"{sid}: its file {'does' if f['sched_own'] else 'does not'} say what a scheduled run may do")
        if f["value_ban"] != ("is never written on a schedule" in s["run"]):
            errs.append(f"{sid}: the rule on values from outside content does not match its file")
        if s["sched"] != (sid in ROUTINE_NAMED):
            errs.append(f"{sid}: named as a routine in the dataset and not in the files, or the reverse")
        if s["sched"] != s["run"].startswith("Named as a routine."):
            errs.append(f"{sid}: its schedule line and its routine mark disagree")
        # readiness keys
        for k in s["core"] + s["adds"] + s["tool"]:
            if k not in SWL:
                errs.append(f"{sid}: unknown readiness key {k}")
        for k in s["must"]:
            if k not in s["adds"] or SWG[k] != "decisions":
                errs.append(f"{sid}: a needed decision must be a decision it asks about")
        for sw, t in (("mail", "email"), ("cal", "calendar")):
            if sw in s["core"] + s["adds"] + s["tool"] and t not in types and "every connected tool" not in types:
                errs.append(f"{sid}: leans on {t}, which is not in its tools table")
        for c in s["cats"]:
            if c not in CATK:
                errs.append(f"{sid}: unknown category {c}")
        # overlap level (our read) against the marking (a fact)
        if s["cls"] == "W" and s["ov"] != 2:
            errs.append(f"{sid}: it changes records, so its overlap level is 2")
        if s["ov"] == 2 and s["cls"] not in "WC":
            errs.append(f"{sid}: level 2 is for skills that change records or write for customers")
        if s["cls"] == "C" and s["ov"] != 2 and sid not in OV_OPEN:
            errs.append(f"{sid}: writes for customers below level 2, and is not listed as awaiting the editor")
        if s["ov"] >= 1 and not (s["cats"] and s["ovnote"] and s["rule"]):
            errs.append(f"{sid}: overlap tier needs a category, a note and a rule")
        if s["ov"] == 0 and s["cats"]:
            errs.append(f"{sid}: no-conflict skill should carry no category")
        if s["seg"] is None and sid != "setup":
            errs.append(f"{sid}: missing motion ratings")
        if s["seg"]:
            for m, (v, fr) in s["seg"].items():
                if v not in D.VALUE or fr not in D.FREQ:
                    errs.append(f"{sid}: bad rating for {m}")
        if set(s["roles"]) != seats:
            errs.append(f"{sid}: bad seats")
    for fact, ids in D.FACTS:
        for i in ids:
            if i not in SK:
                errs.append(f"fact '{fact}' names unknown skill {i}")
    for (fact, sid) in D.FACT_NOTES:
        if sid not in dict(D.FACTS).get(fact, []):
            errs.append(f"a note on '{fact}' for {sid}, which does not use it")
    n_uo = sum(1 for sid, f in FACTS.items() if "update-opportunity" in f["refs"])
    n_none = sum(1 for f in FACTS.values() if f["gated_none"])
    if n_uo != 22:
        errs.append(f"update-opportunity is named by {n_uo} skills, the copy says 22")
    if n_none != 18:
        errs.append(f"{n_none} skills list no write, the copy says 18")
    need = {sid: [t["type"] for t in f["tools"] if t["required"]] for sid, f in FACTS.items()}
    if {k: v for k, v in need.items() if v} != {"call-summary": ["transcripts"], "inbox-sweep": ["email"]}:
        errs.append("the set of 'needed' tools changed: %s" % {k: v for k, v in need.items() if v})
    if len(ROUTINE_NAMED) != 7:
        errs.append("the skills named as routines changed: %s" % ROUTINE_NAMED)
    scoped = sorted(sid for sid, s in SK.items() if s["scope"])
    if scoped != sorted(["weekly-wrap", "deal-signals", "deal-slip-scenario", "pipeline-review", "forecast", "team-pipeline",
                         "rep-context", "win-loss-review", "customer-voice", "renewal-radar"]):
        errs.append("the skills that take a wider scope changed")
    if errs:
        print("BUILD CHECKS FAILED")
        for x in errs:
            print(" -", x)
        sys.exit(1)
    return n_uo, n_none

def num(n, cap=False):
    """Words for one to nine. At the start of a sentence (cap=True), words up to twenty."""
    w = (STARTWORD if cap else NUMWORD).get(n, str(n))
    return w.capitalize() if cap else w


def tagform(k, s=None):
    """The switch's short label as a tag: no leading article, capital first letter. A skill whose file names only
    part of what a switch covers, or more, gives its own words for it."""
    t = re.sub(r"^(a|an) ", "", (s["tagas"].get(k) if s else None) or SWL[k])
    return t[0].upper() + t[1:]


def lean_parts(s):
    extra = [k for k in s["adds"] if k not in s["tool"]]
    return {
        "Built around": list(s["core"]),
        "Needs, or asks once": list(s["must"]),
        "Reads through": list(s["tool"]),
        "Better with": [k for k in extra if SWG[k] != "decisions" or k in s["quiet"]],
        "Takes from your CRM's setup": [k for k in extra if k in s["crmset"]],
        "May ask about": [k for k in extra if SWG[k] == "decisions" and k not in s["must"] and k not in s["quiet"] and k not in s["crmset"]],
    }


def lean_text(s):
    """Plain-text form, used in the table and in skills.json."""
    parts = lean_parts(s)
    out = []
    if s["need"]:
        out.append("Needs: " + s["need"])
    if parts["Built around"]:
        out.append("Built around: " + "; ".join(tagform(k, s).lower() for k in parts["Built around"]) + ".")
    if parts["Needs, or asks once"]:
        out.append("Needs, or asks once: " + "; ".join(tagform(k, s).lower() for k in parts["Needs, or asks once"]) + ".")
    if parts["Reads through"]:
        out.append("Reads through: " + "; ".join(tagform(k, s).lower() for k in parts["Reads through"]) + ". Without that, an export or a paste.")
    else:
        out.append("Needs no tool of its own.")
    if parts["Better with"]:
        out.append("Better with: " + "; ".join(tagform(k, s).lower() for k in parts["Better with"]) + ".")
    if parts["Takes from your CRM's setup"]:
        out.append("Takes from your CRM's setup: " + "; ".join(tagform(k, s).lower() for k in parts["Takes from your CRM's setup"]) + ".")
    if parts["May ask about"]:
        out.append("May ask about: " + "; ".join(tagform(k, s).lower() for k in parts["May ask about"]) + ".")
    text = " ".join(out).replace("crm", "CRM").replace("icp", "ICP")
    return text + (" Note: " + s["leannote"] if s["leannote"] else "")


def lean_html(s):
    parts = lean_parts(s)
    rows = []
    if s["need"]:
        rows.append(f'<div><span class="lbl">Needs</span><span class="fine">{e(s["need"])}</span></div>')
    for label, keys in parts.items():
        if label == "Reads through" and not keys:
            rows.append('<div><span class="lbl">Reads through</span><span class="fine">No tool of its own.</span></div>')
            continue
        if not keys:
            continue
        tags = "".join(f'<span class="tool{" need" if label == "Built around" else ""}">{e(tagform(k, s))}</span>' for k in keys)
        tail = '<span class="fine">or an export or a paste</span>' if label == "Reads through" else ""
        rows.append(f'<div><span class="lbl">{label}</span><span class="chips">{tags}{tail}</span></div>')
    if s["leannote"]:
        rows.append(f'<div><span class="lbl">Note</span><span class="fine">{e(s["leannote"])}</span></div>')
    return '<div class="lean">' + "".join(rows) + "</div>"


def how_often(s):
    """One sentence on frequency: the usual cadence, then the motions that differ."""
    names = {k: n[0].lower() + n[1:] for k, n, *_ in D.MOTIONS}
    fr = {k: f for k, (v, f) in s["seg"].items() if v > 0}
    if not fr:
        return ""
    order = [k for k, *_ in D.MOTIONS]
    counts = {}
    for f in fr.values():
        counts[f] = counts.get(f, 0) + 1
    mode = sorted(counts, key=lambda f: (-counts[f], -f))[0]
    out = [D.FREQ[mode] + "."]
    for f in sorted((x for x in counts if x != mode), reverse=True):
        where = [names[k] for k in order if fr.get(k) == f]
        place = where[0] if len(where) == 1 else (where[0] + ", and in " + where[1] if len(where) == 2 else "; ".join(where))
        out.append(f"{D.FREQ[f]} in {place}.")
    text = " ".join(out)
    if s["id"] in D.EVENT:
        # used when something calls for it: the run card's timing line, then the frequency as a rough cadence
        return D.RUN[s["id"]]["when"] + " As a rough cadence: " + text[0].lower() + text[1:]
    return text

def check_run():
    """Hold the day-to-day layer to the same files: every skill has a run card, a wider scope or a scheduled request
    exists only where the skill's file takes one, and every link in a routine is stated by one of its two files."""
    ids = [s["id"] for s in D.SKILLS]
    assert sorted(D.RUN) == sorted(ids), "run cards: " + str(set(ids) ^ set(D.RUN))
    for sid, r in D.RUN.items():
        assert r["depth"] in D.MODEL_TIERS, sid
        assert not r["team"] or sid in SCOPED, f"{sid}: a request for a wider scope, but the file takes none"
        assert not r["sched"] or sid in SCHED, f"{sid}: a scheduled request, but the file has no schedule"
        assert not r["both"] or (r["team"] and r["sched"]), sid
        assert "[" in r["say"] or r["say"].endswith((".", "?")), sid
        assert len(r["words"]) == 3, sid
    for sid in SCHED:
        assert D.RUN[sid]["sched"], f"{sid}: named as a routine, but no scheduled request"
    for key, _name, _when, roles, paths in D.ROUTINES:
        assert set(roles) <= {k for k, *_ in D.SEATS}, key
        for _label, steps in paths:
            assert steps[0][1] is None and all(st[0] in SK and len(st) in (3, 4) for st in steps), key
            for (a, *_ra), (b, kb, *_rb) in zip(steps, steps[1:]):
                assert kb in (">", "~", "<", "@"), (key, b)
                if kb in (">", "~"):
                    assert b in SK[a]["hand"], f"routine {key}: {a} does not name {b} as a next step in its file"
                else:
                    assert a in SK[b]["draws"], f"routine {key}: {b} does not draw on {a} in its file"
        for k in roles:
            assert any(SK[steps[0][0]]["roles"][k] for _l, steps in paths), f"routine {key}: its first skill is not rated for {k}"
    assert set(D.ROUTINE_NOTES) <= {r[0] for r in D.ROUTINES}
