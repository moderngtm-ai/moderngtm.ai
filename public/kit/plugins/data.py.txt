def A(rec, cust=("no", ""), out=("no", ""), run=""):
    """What a skill can do when you ask, in four facts, each restated by hand from its own file.
    rec  : records it can create or change. "own": its own file lists the write. "via": it lists none and passes
           the change to another skill. "no".
    cust : anything addressed to a customer. "send": it can send an email when asked. "draft": it makes a document
           or a draft and does not send. "no".
    out  : where its own output can go. "push": it can post it to chat or email it. "doc": it keeps a document or
           a page. "no".
    run  : what a scheduled run of it may do, where its own file says so."""
    for part, kinds in ((rec, ("own", "via", "no")), (cust, ("send", "draft", "no")), (out, ("push", "doc", "no"))):
        assert part[0] in kinds, part
    return (rec, cust, out, run)


def reach(act):
    """The furthest a skill goes by its own file. This sets the marking of its name everywhere in the entry."""
    rec, cust, out, _run = act
    return "W" if rec[0] == "own" else "C" if cust[0] != "no" else "P" if out[0] == "push" else "R"


def S(id, group, tag, does, get, ask, act, roles, ov, seg, why,
      cats=(), ovnote="", rule="", sched=False, io=None, hand=(), when=None, draws=(), dwhen=None, scope=None,
      tool=(), core=(), adds=(), must=(), quiet=(), crmset=(), tagas=None, need="", leannote="", rnote="", watch="", says=""):
    """when   : the condition a file puts on a handoff, by skill. The key "*" is one condition for all of them.
    dwhen  : the condition a file puts on a skill it draws on.
    rnote  : what a step of the file reads beyond its tools table, or the condition on a tool it reads.
    quiet  : a decision the skill uses when it is written down and never asks about.
    crmset : a decision the file takes from the CRM's own setup and names no question about.
    tagas  : the words for a switch on this skill's card, where its file names only part of what the switch covers.
    leannote : a source or a fallback the readiness switches have no key for."""
    rec, cust, out, run = act
    # setup is marked apart: with write tools on, its dashboard's tiles carry write actions that run on a click.
    return dict(id=id, group=group, tag=tag, does=does, get=get, ask=list(ask), cls="S" if id == "setup" else reach(act),
                rec=rec, cust=cust, out=out, run=run,
                change=" ".join(t for _k, t in (rec, cust, out) if t),
                roles=roles, ov=ov, cats=list(cats), ovnote=ovnote, rule=rule, sched=sched,
                io=dict(io or {}), hand=list(hand), when=dict(when or {}), draws=list(draws), dwhen=dict(dwhen or {}),
                scope=scope[0] if scope else "", team=bool(scope and scope[1]), rep=bool(scope and scope[2]),
                tool=list(tool), core=list(core), adds=list(adds), must=list(must), quiet=list(quiet),
                crmset=list(crmset), tagas=dict(tagas or {}), need=need,
                leannote=leannote, rnote=rnote, watch=watch, says=says, seg=seg, why=why)


def R(sdr=0, ae=0, am=0, mgr=0, lead=0, ops=0, founder=None):
    # A founder who sells alone takes the highest of the three individual contributor marks, unless one is set here.
    return dict(sdr=sdr, ae=ae, am=am, founder=max(sdr, ae, am) if founder is None else founder, mgr=mgr, lead=lead, ops=ops)
