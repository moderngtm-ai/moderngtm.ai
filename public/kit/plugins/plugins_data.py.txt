def job_slug(name):
    """lead-qualification, crm-updates: the part of an address that names a job."""
    import re
    return re.sub(r"[^a-z0-9]+", "-", name.lower()).strip("-")


def job_lower(name):
    """A job's name inside a sentence: lower case, with CRM kept as it is."""
    return name if name.startswith("CRM") else name[0].lower() + name[1:]
