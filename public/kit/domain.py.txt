# -*- coding: utf-8 -*-
"""Make the domain version of the Modern GTM page, and the files that travel with it at https://moderngtm.ai/.

  python3 domain.py IN_HTML OUT_DIR --llms LLMS_TXT --robots ROBOTS_TXT [--date YYYY-MM-DD]

Runs after merge.py, on the page merge.py writes (the site's or the staging copy's). It changes the page in
seven places and writes the page, robots.txt, llms.txt, sitemap.xml, og.png and fonts/ into OUT_DIR:

  1. Head: fonts served from the site in place of Google Fonts, the canonical address and the preview-image tags.
  2. Contribute: the route for proposals, which replaces "Intake opens with the public site."
  3. Contribute, Reading the site by machine: sitemap.xml and the public repository.
  4. Contribute, Reusing a figure: the line that reuse terms are not set.
  5. About: the privacy notice.
  6. Home: the Latest list moves to a Changes view of its own, reached from the footer and from the
     "Evidence through" date in the header and the footer.
  7. Body: an empty fixed element at the top edge, painted with nothing, whose colour Safari on iPhone reads
     for the strip behind the status bar. Since iOS 26 Safari ignores theme-color and takes that colour from
     the first fixed or sticky element at the top edge, else from the body's background, which is white here.

Each change is anchored on text that must occur exactly once, so a page that has drifted stops the script
rather than producing a half-changed page. The fonts are Archivo and Source Serif 4 (SIL Open Font License 1.1),
from the Fontsource variable packages; their licenses travel in fonts/."""
import datetime
import os
import re
import shutil
import sys

SITE = "https://moderngtm.ai/"
REPO = "https://github.com/moderngtm-ai/moderngtm.ai"
LINKEDIN = "https://www.linkedin.com/in/nathanfierley/"   # the owner's LinkedIn profile; None names LinkedIn without a link
HERE = os.path.dirname(os.path.abspath(__file__))
ASSETS = os.path.join(HERE, "domain_assets")

LATIN = ("U+0000-00FF,U+0131,U+0152-0153,U+02BB-02BC,U+02C6,U+02DA,U+02DC,U+0304,U+0308,U+0329,U+2000-206F,"
         "U+20AC,U+2122,U+2191,U+2193,U+2212,U+2215,U+FEFF,U+FFFD")
LATIN_EXT = ("U+0100-02BA,U+02BD-02C5,U+02C7-02CC,U+02CE-02D7,U+02DD-02FF,U+0304,U+0308,U+0329,U+1D00-1DBF,"
             "U+1E00-1E9F,U+1EF2-1EFF,U+2020,U+20A0-20AB,U+20AD-20C4,U+2113,U+2C60-2C7F,U+A720-A7FF")
FACES = [  # family, style, weight range, extra descriptor, file, unicode range
    ("Archivo", "normal", "100 900", "font-stretch:62% 125%;", "archivo-latin-wdth-normal.woff2", LATIN),
    ("Archivo", "normal", "100 900", "font-stretch:62% 125%;", "archivo-latin-ext-wdth-normal.woff2", LATIN_EXT),
    ("Source Serif 4", "normal", "200 900", "", "source-serif-4-latin-opsz-normal.woff2", LATIN),
    ("Source Serif 4", "normal", "200 900", "", "source-serif-4-latin-ext-opsz-normal.woff2", LATIN_EXT),
    ("Source Serif 4", "italic", "200 900", "", "source-serif-4-latin-opsz-italic.woff2", LATIN),
    ("Source Serif 4", "italic", "200 900", "", "source-serif-4-latin-ext-opsz-italic.woff2", LATIN_EXT),
]
FONT_FILES = [f[4] for f in FACES] + ["OFL-Archivo.txt", "OFL-Source-Serif-4.txt"]

GOOGLE_FONTS = [
    '<link rel="preconnect" href="https://fonts.googleapis.com">',
    '<link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>',
    '<link rel="stylesheet" href="https://fonts.googleapis.com/css2?family=Archivo:wdth,wght@100..125,400..800'
    '&amp;family=Source+Serif+4:ital,opsz,wght@0,8..60,400..700;1,8..60,400..700&amp;display=swap">',
]
def font_head():
    pre = "".join(f'<link rel="preload" href="fonts/{f}" as="font" type="font/woff2" crossorigin>\n'
                  for f in ("archivo-latin-wdth-normal.woff2", "source-serif-4-latin-opsz-normal.woff2"))
    rules = "".join(
        f'@font-face{{font-family:"{fam}";font-style:{style};font-display:swap;font-weight:{wt};{extra}'
        f'src:url(fonts/{fn}) format("woff2");unicode-range:{ur}}}\n'
        for fam, style, wt, extra, fn, ur in FACES)
    links = ".asof a{text-decoration-color:var(--lime)}.asof a:hover{color:var(--lime)}\n"
    # The status bar on iPhone: Safari reads the background-color of a full-width fixed element at the top edge
    # when it is more than 10px tall. Clipped to text it has none of, the element paints nothing; elsewhere it is 0px.
    tint = (".tint{position:fixed;top:0;left:0;right:0;z-index:2147483647;height:0;margin:0;border:0;pointer-events:none;"
            "background-color:var(--v);-webkit-background-clip:text;background-clip:text}\n"
            "@supports (-webkit-touch-callout:none){@media (hover:none) and (pointer:coarse){.tint{height:11px}}}\n")
    return pre + "<style>\n" + rules + links + tint + "</style>"


def meta_head():
    return "\n".join([
        f'<link rel="canonical" href="{SITE}">',
        f'<meta property="og:url" content="{SITE}">',
        '<meta property="og:site_name" content="Modern GTM">',
        f'<meta property="og:image" content="{SITE}og.png">',
        '<meta property="og:image:width" content="1200">',
        '<meta property="og:image:height" content="630">',
        '<meta property="og:image:alt" content="Modern GTM: evidence, plays and tools for revenue teams of people and agents">',
        '<meta name="twitter:card" content="summary_large_image">',
        '<link rel="sitemap" type="application/xml" href="sitemap.xml">',
    ])


def linkedin():
    return f'<a href="{LINKEDIN}">LinkedIn</a>' if LINKEDIN else "LinkedIn"


def changes():
    route = (
        '<p style="margin:16px 0 0"><b>How to send a proposal.</b> File it as an issue in the site’s public GitHub '
        f'repository: <a href="{REPO}/issues/new?template=finding.yml">propose a finding</a> or '
        f'<a href="{REPO}/issues/new?template=correction.yml">report a correction</a>. Each form asks for the fields '
        'above, and an AI agent can file one for the person it acts for. You can also send a proposal to Nathan Fierley '
        f'on {linkedin()}. Issues on GitHub are public.</p>')
    privacy = (
        '<p><b>Privacy.</b> Cloudflare serves this site. It counts visits with its Web Analytics, which sets no cookies '
        'and stores nothing in your browser, and it reports which AI crawlers visit. Nothing on the site identifies a '
        'visitor: it carries no tracking pixels, advertising tags or marketing scripts. A proposal filed on GitHub is '
        'public there, under GitHub’s terms.</p>')
    return [
        ("head: Google Fonts preconnect", GOOGLE_FONTS[0] + "\n", ""),
        ("head: Google Fonts preconnect (static)", GOOGLE_FONTS[1] + "\n", ""),
        ("head: fonts served from the site", GOOGLE_FONTS[2], font_head()),
        ("body: the status bar on iPhone", "<body>\n", '<body>\n<div class="tint" aria-hidden="true"></div>\n'),
        ("head: canonical and preview image", '<meta property="og:type" content="website">',
         '<meta property="og:type" content="website">\n' + meta_head()),
        ("contribute: the route for proposals",
         '<p class="note" style="margin:16px 0 0">Intake opens with the public site. How to send a proposal will be '
         'published here.</p>', route),
        ("contribute: files that travel with the page",
         '<p>The whole site is one page that reads without scripts. Three files travel with it.</p>',
         '<p>The whole site is one page that reads without scripts. Four files travel with it, and every file behind '
         f'the site is public on <a href="{REPO}">GitHub</a>.</p>'),
        ("contribute: sitemap.xml",
         '<li><b>robots.txt</b>: open to search and AI crawlers</li>',
         '<li><b>robots.txt</b>: open to search and AI crawlers</li>\n'
         '<li><b>sitemap.xml</b>: the site’s address for search engines</li>'),
        ("contribute: reuse terms",
         'Cite the original source for the figure and Modern GTM for the grade.</p>',
         'Cite the original source for the figure and Modern GTM for the grade. Terms for reusing the site’s own '
         'text, tables and tools are not set yet, so ask before republishing them. Each figure stays under its '
         'source’s terms.</p>'),
        ("about: privacy notice",
         'and their figures are checked, graded and marked as vendor sources.</p>',
         'and their figures are checked, graded and marked as vendor sources.</p>\n' + privacy),
    ]


CHANGES_LEDE = ("What changed in the register and on the site, newest first. Appendix C of the report logs each "
                "change to the report itself.")


def move_latest(page):
    """Cut the Latest list from Home and give it a view of its own, before the plugin area."""
    head = "<h2>Latest</h2>"
    assert page.count(head) == 1, "Latest: heading found %d times" % page.count(head)
    i = page.index(head)
    start = page.rindex('<section class="sec soft tight">', 0, i)
    end = page.index("</section>", i) + len("</section>")
    tail = "\n</div>\n<!-- RESEARCH -->"
    assert page[end:end + len(tail)] == tail, "Latest is no longer the last section of Home"
    block = page[start:end]
    page = page[:start].rstrip("\n") + page[end:]
    view = ("<!-- CHANGES -->\n<div data-view=\"changes\" id=\"changes\">\n<section class=\"open\">\n<div class=\"wrap\">\n"
            f"<h1>Changes</h1>\n<p class=\"lede\">{CHANGES_LEDE}</p>\n</div>\n</section>\n{block}\n</div>\n")
    anchor = '<div class="plg" id="plg">'
    assert page.count(anchor) == 1, "plugin area anchor"
    page = page.replace(anchor, view + anchor)
    for label, pat, rep in (
            ("header date", r'<span class="asof">(Evidence through [^<]+)</span>', r'<span class="asof"><a href="#changes">\1</a></span>'),
            ("footer date", r'<span>(Evidence through [^<]+)</span></div>', r'<span><a href="#changes">\1</a></span></div>'),
            ("footer link", r'<a href="#contribute">Contribute</a></nav>', r'<a href="#contribute">Contribute</a><a href="#changes">Changes</a></nav>')):
        page, n = re.subn(pat, rep, page)
        assert n == 1, f"{label}: found {n} times"
    return page


def llms(text):
    head = "# Modern GTM\n"
    assert text.startswith(head), "llms.txt: unexpected first line"
    anchor = "- [About and method](/#about)"
    assert text.count(anchor) == 1, "llms.txt: Contributing list changed"
    text = text.replace(anchor, anchor + "\n"
                        "- [Changes](/#changes): what changed in the register and on the site, by date\n"
                        f"- [Propose a finding]({REPO}/issues/new?template=finding.yml): a GitHub issue form\n"
                        f"- [Report a correction]({REPO}/issues/new?template=correction.yml): a GitHub issue form\n"
                        f"- [Files behind the site]({REPO}): the public repository, build kit included")
    return text.replace(head, head + f"\nCanonical address: {SITE}\n", 1)


def main():
    args = sys.argv[1:]
    src, out = args[0], args[1]
    opt = {args[i]: args[i + 1] for i in range(2, len(args) - 1, 2)}
    date = opt.get("--date") or datetime.date.today().isoformat()
    page = open(src, encoding="utf-8").read()
    for label, old, new in changes():
        n = page.count(old)
        assert n == 1, f"{label}: anchor found {n} times"
        page = page.replace(old, new)
    page = move_latest(page)
    for g in GOOGLE_FONTS:
        assert g not in page, "a Google Fonts link is left"
    os.makedirs(os.path.join(out, "fonts"), exist_ok=True)
    with open(os.path.join(out, "index.html"), "w", encoding="utf-8") as f:
        f.write(page)
    robots = open(opt["--robots"], encoding="utf-8").read().rstrip("\n")
    assert "Sitemap:" not in robots
    with open(os.path.join(out, "robots.txt"), "w", encoding="utf-8") as f:
        f.write(robots + f"\n\nSitemap: {SITE}sitemap.xml\n")
    with open(os.path.join(out, "llms.txt"), "w", encoding="utf-8") as f:
        f.write(llms(open(opt["--llms"], encoding="utf-8").read()))
    with open(os.path.join(out, "sitemap.xml"), "w", encoding="utf-8") as f:
        f.write('<?xml version="1.0" encoding="UTF-8"?>\n'
                '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n'
                f'  <url><loc>{SITE}</loc><lastmod>{date}</lastmod></url>\n'
                '</urlset>\n')
    shutil.copyfile(os.path.join(ASSETS, "og.png"), os.path.join(out, "og.png"))
    for fn in FONT_FILES:
        shutil.copyfile(os.path.join(ASSETS, "fonts", fn), os.path.join(out, "fonts", fn))
    print(f"domain version written to {out}: index.html, robots.txt, llms.txt, sitemap.xml, og.png, "
          f"fonts/ ({len(FONT_FILES)} files); LinkedIn link {'set' if LINKEDIN else 'not set'}")


if __name__ == "__main__":
    main()
